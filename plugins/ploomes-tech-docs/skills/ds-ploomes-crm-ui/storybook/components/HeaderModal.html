export const AllVariants: Story = {
  render: () => (
    <div style={{ display: 'flex', flexDirection: 'column', gap: 2 }}>
      {[
        { label: 'null (no left)', props: {} },
        {
          label: 'text',
          props: { left: 'text' as const, title: 'Título do modal' },
        },
        {
          label: 'button_soft',
          props: {
            left: 'button_soft' as const,
            leftButtons: [{ label: 'Lorem Ipsum' }],
          },
        },
        {
          label: '2_button_soft',
          props: {
            left: '2_button_soft' as const,
            leftButtons: [{ label: 'Lorem Ipsum' }, { label: 'Lorem Ipsum' }],
          },
        },
        {
          label: 'breadcrumbs',
          props: {
            left: 'breadcrumbs' as const,
            breadcrumbItems: [
              { label: 'Home', href: '#' },
              { label: 'Seção', href: '#' },
              { label: 'Página' },
            ],
            breadcrumbSize: 'sm' as const,
          },
        },
        {
          label: 'text + intercomButton',
          props: {
            left: 'text' as const,
            title: 'Com ajuda',
            intercomButton: true,
          },
        },
      ].map(({ label, props }) => (
        <div key={label}>
          <p
            style={{
              fontSize: 11,
              color: '#8b8595',
              padding: '4px 8px',
              background: '#f5f3f8',
            }}
          >
            {label}
          </p>
          <HeaderModal {...props} onClose={() => {}} />
        </div>
      ))}
    </div>
  ),
}
