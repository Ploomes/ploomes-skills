export const AllSizes: Story = {
  render: () => (
    <div style={{ display: 'flex', flexDirection: 'column', gap: 16 }}>
      {(['small', 'normal', 'big', 'bigger'] as const).map((size) => (
        <Collapse
          key={size}
          size={size}
          title="Lorem Ipsum"
          subtitle="Subtitle"
          icon={
            <BoxIcon
              size={
                size === 'small'
                  ? 12
                  : size === 'normal'
                    ? 16
                    : size === 'big'
                      ? 22
                      : 28
              }
            />
          }
          rightInfo={<Badge color="confirmation">{size}</Badge>}
          defaultOpen={size === 'big'}
        >
          <SampleContent />
        </Collapse>
      ))}
    </div>
  ),
}
