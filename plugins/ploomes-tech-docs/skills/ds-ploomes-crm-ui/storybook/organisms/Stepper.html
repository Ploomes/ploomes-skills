export const ThreeSteps: Story = {
  name: '3 steps',
  render: () => (
    <div className="flex flex-col gap-[15px]">
      {['Início', 'Processamento', 'Conclusão'].map((_, active, arr) => {
        const steps = arr.map((__, i) => ({
          id: i,
          label: ['Início', 'Processamento', 'Conclusão'][i],
        }));
        return (
          <Stepper
            key={active}
            steps={steps}
            activeStep={active}
            color="brand"
          />
        );
      })}
    </div>
  ),
}

      { id: 2, label: 'Fim' },

      </div>
    );
  },
