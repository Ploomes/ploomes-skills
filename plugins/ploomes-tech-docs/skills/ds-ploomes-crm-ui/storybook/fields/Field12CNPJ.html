Receita Federal Flow

fields-field12cnpj--receita-federal-flow · export ReceitaFederalFlow
código da story
export const ReceitaFederalFlow: Story = {
  render: function Render(args) {
    const [active, setActive] = useState(false);
    const [busy, setBusy] = useState(false);
    return (
      <Field12CNPJ
        {...args}
        defaultValue="11222333000181"
        receitaFederalActive={active}
        receitaFederalLoading={busy}
        onReceitaFederalClick={() => {
          if (active) {
            setActive(false);
            return;
          }
          setBusy(true);
          setTimeout(() => {
            setBusy(false);
            setActive(true);
          }, 1500);
        }}
      />
    );
  },
}
