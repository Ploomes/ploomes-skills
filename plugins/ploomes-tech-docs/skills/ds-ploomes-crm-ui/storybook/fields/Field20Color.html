export const Controlled: Story = {
  render: function Render(args) {
    const [color, setColor] = useState<string | null>('#7E33E6');
    return (
      <div style={{ display: 'flex', flexDirection: 'column', gap: 8 }}>
        <Field20Color
          {...args}

          Cor: {color ?? 'null'}
        </span>
      </div>
    );
  },
