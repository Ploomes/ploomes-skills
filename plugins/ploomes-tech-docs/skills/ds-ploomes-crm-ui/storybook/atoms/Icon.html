export const AllVariants: Story = {
  render: () => {
    const colors = [
      'default',
      'brand',
      'success',
      'danger',
      'alert',
      'confirmation',
    ] as const;
    const sizes = [
      'tiny_10',
      'small_15',
      'normal_20',
      'big_30',
      'giant_90',
    ] as const;
    return (
      <div style={{ display: 'flex', flexDirection: 'column', gap: 24 }}>
        {colors.map((color) => (
          <div
            key={color}
            style={{ display: 'flex', gap: 16, alignItems: 'flex-end' }}
          >
            {sizes.map((size) => (
              <Icon key={size} icon={House} color={color} size={size} />
            ))}
            <span style={{ fontSize: 11, color: '#888', alignSelf: 'center' }}>
              {color}
            </span>
          </div>
        ))}
      </div>
    );
  },
}
