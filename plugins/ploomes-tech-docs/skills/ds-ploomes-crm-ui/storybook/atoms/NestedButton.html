export const AllVariants: Story = {
  render: () => {
    const sizes = ['small_20', 'normal_25', 'big_30'] as const;
    const iconSize = {
      small_20: 'tiny_10',
      normal_25: 'small_15',
      big_30: 'small_15',
    } as const;
    return (
      <div style={{ display: 'flex', flexDirection: 'column', gap: 24 }}>
        <div
          style={{
            display: 'flex',
            gap: 8,
            alignItems: 'center',
            flexWrap: 'wrap',
          }}
        >
          {sizes.map((size) => (
            <React.Fragment key={size}>
              <NestedButton
                size={size}
                mode="light"
                outline
                aria-label="Mais opções"
              >
                <Icon icon={Ellipsis} size={iconSize[size]} />
              </NestedButton>
              <NestedButton
                size={size}
                mode="light"
                outline={false}
                aria-label="Mais opções"
              >
                <Icon icon={Ellipsis} size={iconSize[size]} />
              </NestedButton>
              <NestedButton
                size={size}
                mode="light"
                outline
                disabled
                aria-label="Mais opções"
              >
                <Icon icon={Ellipsis} size={iconSize[size]} />
              </NestedButton>
            </React.Fragment>
          ))}
        </div>
        <div
          style={{
            display: 'flex',
            gap: 8,
            alignItems: 'center',
            flexWrap: 'wrap',
            background: '#1B102C',
            padding: 12,
            borderRadius: 8,
          }}
        >
          {sizes.map((size) => (
            <React.Fragment key={size}>
              <NestedButton
                size={size}
                mode="dark"
                outline
                aria-label="Mais opções"
              >
                <Icon icon={Ellipsis} size={iconSize[size]} />
              </NestedButton>
              <NestedButton
                size={size}
                mode="dark"
                outline={false}
                aria-label="Mais opções"
              >
                <Icon icon={Ellipsis} size={iconSize[size]} />
              </NestedButton>
              <NestedButton
                size={size}
                mode="dark"
                outline
                disabled
                aria-label="Mais opções"
              >
                <Icon icon={Ellipsis} size={iconSize[size]} />
              </NestedButton>
            </React.Fragment>
          ))}
        </div>
      </div>
    );
  },
}
